# 机器学习
import numpy as np
import pandas as pd

from sklearn.base import BaseEstimator, TransformerMixin
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.linear_model import LinearRegression, SGDRegressor
from sklearn.metrics import root_mean_squared_error
from sklearn.model_selection import StratifiedShuffleSplit, cross_val_score, GridSearchCV
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.tree import DecisionTreeRegressor
from sklearn.ensemble import RandomForestRegressor

def load_data(path: str) -> pd.DataFrame:
    """加载数据并创建收入分箱列（仅用于分层采样）"""
    housing = pd.read_csv(path)
    housing["income_cat"] = pd.cut(
        housing["median_income"],
        bins=[0.0, 1.5, 3.0, 4.5, 6.0, np.inf],
        labels=[1, 2, 3, 4, 5]
    )
    return housing

def stratified_split(df: pd.DataFrame, test_size=0.2, random_state=42):
    """基于 income_cat 进行分层抽样"""
    split = StratifiedShuffleSplit(n_splits=1, test_size=test_size, random_state=random_state)
    for train_index, test_index in split.split(df, df["income_cat"]):
        train_set = df.iloc[train_index].copy()
        test_set = df.iloc[test_index].copy()
    return train_set, test_set

def build_preprocessing_pipeline(num_attribs: list[str], cat_attribs: list[str]):
    """构建包含自定义特征工程的预处理流水线"""

    class CombinedAttributesAdder(BaseEstimator, TransformerMixin):
        def __init__(self, add_bedrooms_per_room=True):
            self.add_bedrooms_per_room = add_bedrooms_per_room

        def fit(self, X, y=None):
            return self

        def transform(self, X):
            # ⚠️ 注意：这些索引严格依赖 num_attribs 的传入顺序
            # 当前顺序: longitude(0), latitude(1), housing_median_age(2),
            #           total_rooms(3), total_bedrooms(4), population(5),
            #           households(6), median_income(7)
            rooms_ix = 3
            bedrooms_ix = 4
            population_ix = 5
            households_ix = 6

            rooms_per_household = X[:, rooms_ix] / X[:, households_ix]
            population_per_household = X[:, population_ix] / X[:, households_ix]

            if self.add_bedrooms_per_room:
                bedrooms_per_room = X[:, bedrooms_ix] / X[:, rooms_ix]
                return np.c_[X, rooms_per_household, population_per_household, bedrooms_per_room]

            return np.c_[X, rooms_per_household, population_per_household]

    full_pipeline = ColumnTransformer([
        ("num", Pipeline([
            ("imputer", SimpleImputer(strategy="median")),
            ("attribs_adder", CombinedAttributesAdder()),
            ("scaler", StandardScaler())
        ]), num_attribs),
        ("cat", OneHotEncoder(handle_unknown="ignore"), cat_attribs),
    ])

    return full_pipeline

def prepare_train_test_data(df: pd.DataFrame):
    """
    完成分层分割 + 预处理，返回准备好的训练/测试特征与标签
    ✅ 已修复：同时删除 median_house_value 和 income_cat
    """
    train_set, test_set = stratified_split(df)

    # 【关键修复】必须同时删除标签列和分层采样辅助列
    X_train = train_set.drop(["median_house_value", "income_cat"], axis=1)
    y_train = train_set["median_house_value"].copy()

    X_test = test_set.drop(["median_house_value", "income_cat"], axis=1)
    y_test = test_set["median_house_value"].copy()

    num_attribs = X_train.drop("ocean_proximity", axis=1).columns.tolist()
    cat_attribs = ["ocean_proximity"]

    full_pipeline = build_preprocessing_pipeline(num_attribs, cat_attribs)

    X_train_prepared = full_pipeline.fit_transform(X_train)
    X_test_prepared = full_pipeline.transform(X_test)  # ✅ 测试集只用 transform

    return X_train_prepared, X_test_prepared, y_train, y_test

def train_and_rmse(model, X_train, y_train, X_test, y_test, model_name="模型"):
    """训练模型并返回测试集 RMSE"""
    model.fit(X_train, y_train)
    rmse = root_mean_squared_error(y_test, model.predict(X_test))
    print(f"{model_name} 的测试集 RMSE: {rmse:.2f}")
    return model, rmse

if __name__ == "__main__":
    housing = load_data("housing.csv")

    X_train_prepared, X_test_prepared, y_train, y_test = prepare_train_test_data(housing)

    # 1. 标准方程线性回归
    model1, rmse1 = train_and_rmse(
        LinearRegression(),
        X_train_prepared, y_train, X_test_prepared, y_test,
        "标准方程线性回归模型"
    )

    # 2. SGD 线性回归
    # ✅ 已修复：关闭正则化、固定随机种子、增加迭代次数，保证与 LinearRegression 公平对比
    model2, rmse2 = train_and_rmse(
        SGDRegressor(penalty=None, max_iter=1000, tol=1e-3, random_state=42),
        X_train_prepared, y_train, X_test_prepared, y_test,
        "随机梯度下降线性回归模型"
    )

    print("\n===== 最终结果 =====")
    print(f"标准方程线性回归模型 RMSE: {rmse1:.2f}")
    print(f"随机梯度下降线性回归模型 RMSE: {rmse2:.2f}")
