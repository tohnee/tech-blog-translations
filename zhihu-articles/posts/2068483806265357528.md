### 🎨 4. 生产级公共函数库

路径：`scripts/common.sh`

所有脚本的基础设施，提供结构化日志、覆盖式进度、进度条、端口管理等能力。

```text
cat > scripts/common.sh << 'EOF'
#!/bin/bash
# ============================================================
# common.sh - 生产级公共函数库
# 提供: 结构化日志 | 覆盖式进度 | 进度条 | 端口管理 | 环境加载
# ============================================================

set -euo pipefail

# ======================== 颜色与图标 ========================
readonly C_RED='\033[0;31m'
readonly C_GREEN='\033[0;32m'
readonly C_YELLOW='\033[1;33m'
readonly C_BLUE='\033[0;34m'
readonly C_CYAN='\033[0;36m'
readonly C_BOLD='\033[1m'
readonly C_NC='\033[0m'

# ======================== 日志系统 ========================
_log() {
    local level="$1" color="$2" icon="$3" msg="$4"
    printf "${color}%s [%-5s] %s${C_NC}\n" "$icon" "$level" "$msg"
}

log_info()    { _log "INFO"  "$C_GREEN"  "✔" "$*"; }
log_warn()    { _log "WARN"  "$C_YELLOW" "⚠" "$*" >&2; }
log_error()   { _log "ERROR" "$C_RED"    "✘" "$*" >&2; }
log_step()    { printf "${C_BOLD}${C_CYAN}➤ %s${C_NC}\n" "$*"; }
log_header()  { printf "\n${C_BOLD}${C_CYAN}━━━ %s ━━━${C_NC}\n" "$*"; }
die()         { log_error "$*"; exit 1; }

# ======================== 覆盖式进度显示 ========================
# ✅ 修复: 使用 if/fi 替代 && ，避免 set -e 下条件为 false 时静默退出
progress_inline() {
    local label="$1" current="$2" total="$3"
    local pct=$((current * 100 / total))
    printf "\r${C_GREEN}✔${C_NC} %s [${C_BOLD}%d/%d${C_NC}] (%d%%)" \
        "$label" "$current" "$total" "$pct"
    if [ "$current" -eq "$total" ]; then echo ""; fi
}

# ======================== 整体进度条 ========================
# ✅ 修复: 同上
progress_bar() {
    local current="$1" total="$2" label="${3:-进度}"
    local width=30
    local pct=$((current * 100 / total))
    local filled=$((current * width / total))
    local empty=$((width - filled))
    local bar=""
    for ((i=0; i<filled; i++)); do bar+="█"; done
    for ((i=0; i<empty; i++)); do bar+="░"; done
    printf "\r${C_CYAN}%s${C_NC} |${C_GREEN}%s${C_NC}| %d%% (%d/%d)" \
        "$label" "$bar" "$pct" "$current" "$total"
    if [ "$current" -eq "$total" ]; then echo ""; fi
}

# ======================== 环境加载与校验 ========================
load_env() {
    [ -n "${SFT_ROOT:-}" ] || die "没有SFT训练根路径"

    local env_file="${SFT_ROOT}/config/settings.env"
    [ -f "$env_file" ] || die "配置文件未找到: $env_file"

    source "$env_file"

    # ✅ 修复: 正向检查 + || die，避免 && die 的 set -e 陷阱
    [ -n "${VERSION:-}" ] || die "VERSION 未在 settings.env 中设置"
    log_info "当前版本: ${VERSION} | 模型: ${MODEL_NAME}"

    for d in "$DATASET_DIR" "$LORA_ADAPTER_DIR" "$MERGED_MODEL_DIR" "$LOG_DIR" "$GGUF_DIR"; do
        [ -d "$d" ] || mkdir -p "$d" || die "无法创建目录: $d"
    done

    for var in MODEL_DIR LLAMA_FACTORY LLAMA_CPP LLAMA_CLI; do
        [ -n "${!var:-}" ] || die "环境变量 $var 未设置"
    done
}

# ======================== 工具函数 ========================
check_cmd()  { command -v "$1" >/dev/null 2>&1 || die "命令未找到: $1"; }
check_file() { [ -f "$1" ] || die "文件不存在: $1"; }
check_dir()  { [ -d "$1" ] || die "目录不存在: $1"; }

kill_port() {
    local port="$1" max_wait="${2:-5}" pid
    pid=$(lsof -t -i:"$port" 2>/dev/null || true)
    if [ -n "$pid" ]; then
        log_warn "端口 $port 被占用 (PID: $pid)，正在清理..."
        kill -9 "$pid" 2>/dev/null || true
        for ((i=1; i<=max_wait; i++)); do
            sleep 1
            pid=$(lsof -t -i:"$port" 2>/dev/null || true)
            if [ -z "$pid" ]; then
                log_info "端口 $port 已释放"
                return 0
            fi
        done
        die "端口 $port 清理超时"
    fi
}

wait_for_server() {
    local url="$1" timeout="${2:-60}" name="${3:-服务}"
    log_step "等待${name}就绪..."
    for ((i=1; i<=timeout; i++)); do
        if curl -sf "$url" > /dev/null 2>&1; then
            log_info "${name}已就绪 (${i}s)"
            return 0
        fi
        progress_inline "等待${name}" "$i" "$timeout"
        sleep 1
    done
    echo ""
    die "${name}启动超时 (${timeout}s)"
}

script_start() { log_header "▶ $(basename "$0")"; }
script_end()   { log_info "◼ $(basename "$0") 完成"; }
EOF
```

```text
cat > scripts/01_convert_base_to_gguf.sh << 'EOF'
#!/bin/bash

set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
source "${SCRIPT_DIR}/common.sh"
load_env
script_start

log_step "1/1 转换基座模型为 GGUF"
if [ -f "$BASE_GGUF" ]; then
    log_info "基座 GGUF 已存在，跳过: $BASE_GGUF"
else
    check_file "$MODEL_DIR/config.json"
    CONVERT_SCRIPT="${LLAMA_CPP}/convert-hf-to-gguf.py"
    [ ! -f "$CONVERT_SCRIPT" ] && CONVERT_SCRIPT="${LLAMA_CPP}/convert_hf_to_gguf.py"
    [ -f "$CONVERT_SCRIPT" ] || die "找不到 llama.cpp 转换脚本"

    log_info "转换中: $MODEL_DIR → $BASE_GGUF"
    
    # ✅ 使用 tee 同时输出到终端和日志文件
    # ✅ PYTHONUNBUFFERED=1 强制 Python 禁用输出缓冲，确保实时显示
    PYTHONUNBUFFERED=1 python3 "$CONVERT_SCRIPT" "$MODEL_DIR" \
        --outfile "$BASE_GGUF" \
        --outtype auto \
        2>&1 | tee "${LOG_DIR}/convert_base.log" || die "基座转换失败，查看: ${LOG_DIR}/convert_base.log"
    
    log_info "转换成功: $BASE_GGUF"
fi

script_end
EOF
```

### 5.3 基座模型跑 prompt

路径：`scripts/02_run_baseline.sh`

```text
cat > scripts/02_run_baseline.sh << 'EOF'
#!/bin/bash

set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
source "${SCRIPT_DIR}/common.sh"
load_env
script_start

check_file "$PROMPTS_FILE"
check_file "$BASE_GGUF"
check_cmd curl; check_cmd jq

SERVER_PORT="${SERVER_PORT:-8080}"
SERVER_HOST="127.0.0.1"

log_step "1/3 启动基座模型服务"
kill_port "$SERVER_PORT"
nohup "$LLAMA_SERVER" \
    -m "$BASE_GGUF" \
    --port "$SERVER_PORT" \
    --host "$SERVER_HOST" \
    > "${LOG_DIR}/baseline_server.log" 2>&1 &
SERVER_PID=$!
wait_for_server "http://${SERVER_HOST}:${SERVER_PORT}/v1/models" 180 "基座服务"

log_step "2/3 执行基准测试"
TOTAL_PROMPTS=$(grep -cve '^\s*$' "$PROMPTS_FILE" || echo 0)
CURRENT=0
> "$BASELINE_RESULTS"

while IFS= read -r prompt || [ -n "$prompt" ]; do
    [ -z "$prompt" ] && continue
    CURRENT=$((CURRENT + 1))
    progress_inline "基准测试" "$CURRENT" "$TOTAL_PROMPTS"

    JSON_DATA=$(jq -n --arg p "$prompt" \
        --argjson n "$INFER_N_PREDICT" --argjson t "$INFER_TEMPERATURE" \
        --argjson tp "$INFER_TOP_P" --argjson rp "$INFER_REPEAT_PENALTY" \
        --argjson s "$INFER_SEED" \
        '{prompt:$p,n_predict:$n,temperature:$t,top_p:$tp,repeat_penalty:$rp,seed:$s}')

    CONTENT=$(curl -sf -X POST "http://${SERVER_HOST}:${SERVER_PORT}/completion" \
        -H "Content-Type: application/json" -d "$JSON_DATA" \
        | jq -r '.content // "ERROR"' 2>/dev/null || echo "ERROR")

    { echo "=== Prompt $CURRENT: $prompt ==="; echo -e "$CONTENT\n"; } >> "$BASELINE_RESULTS"
done < "$PROMPTS_FILE"

log_step "3/3 清理资源"
kill "$SERVER_PID" 2>/dev/null; wait "$SERVER_PID" 2>/dev/null || true
log_info "结果已保存: $BASELINE_RESULTS ($CURRENT 条)"

script_end
```

### 5.5 通过训练的模型跑 prompt

路径：`scripts/04_evaluate_lora.sh`

```text
cat > scripts/04_evaluate_lora.sh << 'EOF'
#!/bin/bash

set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
source "${SCRIPT_DIR}/common.sh"
load_env
script_start

check_file "$PROMPTS_FILE"
check_file "$LORA_ADAPTER_DIR/adapter_config.json"
check_cmd llamafactory-cli; check_cmd curl; check_cmd jq

SERVER_PORT=8000
SERVER_HOST="127.0.0.1"

# ─── 中断安全：确保异常退出时 JSON 文件仍能闭合 ───
_cleanup_json() {
    if [ -f "$LORA_RESULT_FILE" ]; then
        # 检查文件是否已正确闭合，未闭合则补全
        if ! jq empty "$LORA_RESULT_FILE" 2>/dev/null; then
            echo '' >> "$LORA_RESULT_FILE"
            echo ']' >> "$LORA_RESULT_FILE"
            log_warn "检测到异常中断，已自动补全 JSON 数组"
        fi
    fi
}
trap _cleanup_json EXIT

log_step "1/3 启动 LoRA API 服务"
kill_port "$SERVER_PORT"
nohup llamafactory-cli api \
    --model_name_or_path "$MODEL_DIR" \
    --adapter_name_or_path "$LORA_ADAPTER_DIR" \
    --template qwen \
    > "${LOG_DIR}/lora_eval_server.log" 2>&1 &
SERVER_PID=$!
wait_for_server "http://${SERVER_HOST}:${SERVER_PORT}/v1/models" 60 "LoRA API"

log_step "2/3 执行 LoRA 评估"
TOTAL_PROMPTS=$(grep -cve '^\s*$' "$PROMPTS_FILE" || echo 0)
CURRENT=0; SUCCESS=0

# 初始化 JSON 数组
echo '[' > "$LORA_RESULT_FILE"
FIRST=true

while IFS= read -r prompt || [ -n "$prompt" ]; do
    [ -z "$prompt" ] && continue
    CURRENT=$((CURRENT + 1))
    progress_inline "LoRA评估" "$CURRENT" "$TOTAL_PROMPTS"

    JSON=$(jq -n --arg c "$prompt" \
        '{model:"default",messages:[{role:"user",content:$c}],temperature:0.7,top_p:0.9,max_tokens:128,stream:false}')

    CONTENT=$(curl -sf -X POST "http://${SERVER_HOST}:${SERVER_PORT}/v1/chat/completions" \
        -H "Content-Type: application/json" -d "$JSON" \
        | jq -r '.choices[0].message.content // "ERROR"' 2>/dev/null || echo "ERROR")

    # 安全构建 JSON 对象并追加（jq --arg 自动处理所有特殊字符转义）
    if [ "$FIRST" = true ]; then
        FIRST=false
    else
        echo ',' >> "$LORA_RESULT_FILE"
    fi
    jq -n --arg p "$prompt" --arg o "$CONTENT" \
        '{prompt: $p, output: $o}' >> "$LORA_RESULT_FILE"

    [ "$CONTENT" != "ERROR" ] && [ -n "$CONTENT" ] && SUCCESS=$((SUCCESS + 1))
done < "$PROMPTS_FILE"

# 正常结束，闭合 JSON 数组
echo '' >> "$LORA_RESULT_FILE"
echo ']' >> "$LORA_RESULT_FILE"

# 移除 EXIT trap（文件已合法，无需再触发补全逻辑）
trap - EXIT

# 校验输出合法性
jq empty "$LORA_RESULT_FILE" || die "结果文件不是合法 JSON: $LORA_RESULT_FILE"

log_step "3/3 清理资源"
kill "$SERVER_PID" 2>/dev/null; wait "$SERVER_PID" 2>/dev/null || true
log_info "评估完成: $SUCCESS/$CURRENT 成功 → $LORA_RESULT_FILE"

script_end
EOF
```

### 5.9 主控流水线

路径：`scripts/99_main_pipeline.sh`

```text
cat > scripts/99_main_pipeline.sh << 'EOF'
#!/bin/bash
# ============================================================
# 99_main_pipeline.sh - SFT 全流程主控（带整体进度条）
# 用法: bash 99_main_pipeline.sh [--skip-sft] [--skip-eval] ...
# ============================================================

set -euo pipefail

SKIP_BASELINE=false; SKIP_CONVERT=false; SKIP_SFT=false
SKIP_EVAL=false; SKIP_MERGE=false; SKIP_FINETUNED=false

while [ $# -gt 0 ]; do
    case "$1" in
        --skip-baseline)  SKIP_BASELINE=true ;;
        --skip-convert)   SKIP_CONVERT=true ;;
        --skip-sft)       SKIP_SFT=true ;;
        --skip-eval)      SKIP_EVAL=true ;;
        --skip-merge)     SKIP_MERGE=true ;;
        --skip-finetuned) SKIP_FINETUNED=true ;;
        *) echo "未知选项: $1"; exit 1 ;;
    esac
    shift
done

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
source "${SCRIPT_DIR}/common.sh"
load_env

# ==================== 定义流水线步骤 ====================
declare -a STEPS=()
declare -a STEP_NAMES=()
declare -a STEP_FLAGS=()

add_step() { STEPS+=("$1"); STEP_NAMES+=("$2"); STEP_FLAGS+=("$3"); }

add_step "01_convert_base_to_gguf.sh"      "转换基座GGUF"   "$SKIP_CONVERT"
add_step "02_run_baseline.sh"              "基座基准测试"   "$SKIP_BASELINE"
add_step "03_run_sft.sh"                   "SFT 训练"      "$SKIP_SFT"
add_step "04_evaluate_lora.sh"             "LoRA 评估"     "$SKIP_EVAL"
add_step "05_merge_lora.sh"                "合并 LoRA"     "$SKIP_MERGE"
add_step "06_convert_finetuned_to_gguf.sh" "转换微调GGUF"   "$SKIP_CONVERT"
add_step "07_run_finetuned.sh"             "微调后测试"     "$SKIP_FINETUNED"

TOTAL_STEPS=${#STEPS[@]}
COMPLETED=0
TOTAL_START=$(date +%s)

log_header "🚀 SFT 全流程开始 | 版本: $VERSION"

for ((i=0; i<TOTAL_STEPS; i++)); do
    script="${STEPS[$i]}"
    name="${STEP_NAMES[$i]}"
    skip="${STEP_FLAGS[$i]}"

    progress_bar "$COMPLETED" "$TOTAL_STEPS" "流水线"

    if [ "$skip" = true ]; then
        log_warn "⏭ 跳过: $name"
    else
        log_step "[$((i+1))/$TOTAL_STEPS] $name"
        if ! bash "${SCRIPT_DIR}/${script}"; then
            die "❌ 步骤失败: $name (${script})"
        fi
    fi
    COMPLETED=$((COMPLETED + 1))
done

progress_bar "$TOTAL_STEPS" "$TOTAL_STEPS" "流水线"

DURATION=$(( $(date +%s) - TOTAL_START ))
log_header "✅ SFT 全流程完成 | 版本: $VERSION"
log_info "总耗时: $((DURATION/60))分$((DURATION%60))秒"
echo ""
log_info "📊 结果文件:"
echo "   基座输出:   $BASELINE_RESULTS"
echo "   LoRA输出:   $LORA_RESULT_FILE"
echo "   微调输出:   $FINETUNED_RESULT_FILE"
EOF
```
